# Keep all compiler and sanitizer settings local to these targets.
function(voice_target_options target)
	set_target_properties(${target} PROPERTIES CXX_EXTENSIONS OFF)
	if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
		target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror)
		if(VOICE_THREAD_SANITIZER)
			target_compile_options(${target} PRIVATE -fsanitize=thread -fno-omit-frame-pointer)
			target_link_options(${target} PRIVATE -fsanitize=thread)
		endif()
		if(VOICE_SANITIZERS)
			target_compile_options(${target} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
			target_link_options(${target} PRIVATE -fsanitize=address,undefined)
		endif()
	endif()
endfunction()

# Exactly one DSP library, shared by tests, preview and future server adapters.
# No Qt, Opus, Python, WAV I/O or Murmur objects in its dependency graph.
add_library(funk-voice-dsp STATIC src/VoiceProcessor.cpp)
target_include_directories(funk-voice-dsp PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/include")
target_compile_features(funk-voice-dsp PUBLIC cxx_std_17)
voice_target_options(funk-voice-dsp)

if(VOICE_BUILD_TESTS)
	enable_testing()
	add_executable(voice-dsp-test tests/dsp_test.cpp)
	target_link_libraries(voice-dsp-test PRIVATE funk-voice-dsp)
	voice_target_options(voice-dsp-test)
	add_test(NAME VoiceDsp COMMAND voice-dsp-test)
endif()

if(VOICE_BUILD_LIVE OR VOICE_BUILD_PREVIEW)
	find_path(OPUS_INCLUDE_DIR opus.h PATH_SUFFIXES opus)
	find_library(OPUS_LIBRARY NAMES opus)
	if(NOT OPUS_INCLUDE_DIR OR NOT OPUS_LIBRARY)
		message(FATAL_ERROR "Analog voice Opus adapters require libopus headers and library")
	endif()
endif()

if(VOICE_BUILD_LIVE)
	add_library(funk-voice-stream STATIC src/AnalogVoiceStream.cpp)
	target_include_directories(funk-voice-stream PUBLIC "${OPUS_INCLUDE_DIR}")
	target_link_libraries(funk-voice-stream PUBLIC funk-voice-dsp "${OPUS_LIBRARY}")
	voice_target_options(funk-voice-stream)
	if(VOICE_BUILD_TESTS)
		add_executable(voice-stream-test tests/stream_test.cpp)
		target_link_libraries(voice-stream-test PRIVATE funk-voice-stream)
		voice_target_options(voice-stream-test)
		if(CMAKE_SYSTEM_NAME STREQUAL "Linux" AND CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
			target_compile_definitions(voice-stream-test PRIVATE VOICE_TEST_CODEC_FAULTS)
			target_link_options(voice-stream-test PRIVATE
				-Wl,--wrap=opus_encode_float -Wl,--wrap=opus_decode_float)
		endif()
		add_test(NAME VoiceStream COMMAND voice-stream-test)
	endif()
endif()
